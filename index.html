الاستجابة تتضمن الحقول decision، risk_score، وlatency_ms لقياس الأداء لحظيًا.
radae_client.py — zsh
import requests

url = "https://api.radae.sa/v1/transactions/verify"

headers = {
    "Authorization": "Bearer RADAE_SECRET_KEY",
    "Content-Type": "application/json"
}

payload = {
    "transaction_id": "txn_98213ZP",
    "account_id": "acc_44120",
    "amount": 15000.00,
    "currency": "SAR",
    "channel": "mobile_app",
    "device_fingerprint": "fp_x91a..."
}

# إرسال العملية إلى محرك التحليل السلوكي لرادع
response = requests.post(url, json=payload, headers=headers)
result = response.json()

print(result["decision"])      # "approved" | "blocked" | "review"
print(result["risk_score"])   # 0 → 100
print(result["latency_ms"])   # 95
